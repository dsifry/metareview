+func TestMatchesExcludeAndAny(t *testing.T) {
+	cases := []struct {
+		file, exclude string
+		want          bool
+	}{
+		{"a/b.go", ".", false},        // "." exclude never matches
+		{"a/b.go", "", false},         // empty exclude never matches

+		if got := matchesExclude(c.file, c.exclude); got != c.want {
